SHOW VARIABLES LIKE  '%slow_query_log%'; # 是否开启慢查询日志

SHOW VARIABLES LIKE  '%long_query_time%'; # 慢查询时间阈值 默认10s
或者
show global variables like 'long_query_time';

set global long_query_time=3; #设置慢查询阈值 （只会记录大于该时间阈值的sql而不是大于等于）

show global status like '%Slow_queries%'; # 查询当前系统中有多少条慢查询记录
